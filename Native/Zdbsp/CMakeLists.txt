cmake_minimum_required(VERSION 3.15)
project(MobileUDBNodes LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_subdirectory(upstream/zlib EXCLUDE_FROM_ALL)
set(SOURCES blockmapbuilder.cpp processor.cpp processor_udmf.cpp sc_man.cpp wad.cpp
  nodebuild.cpp nodebuild_events.cpp nodebuild_extract.cpp nodebuild_gl.cpp
  nodebuild_utility.cpp nodebuild_classify_nosse2.cpp)
list(TRANSFORM SOURCES PREPEND "upstream/")
if(IOS)
  add_library(mobileudb_nodes STATIC bridge.cpp ${SOURCES})
else()
  add_library(mobileudb_nodes SHARED bridge.cpp ${SOURCES})
endif()
target_compile_definitions(mobileudb_nodes PRIVATE DISABLE_SSE NO_MAP_VIEWER)
target_include_directories(mobileudb_nodes PRIVATE upstream upstream/zlib ${CMAKE_CURRENT_BINARY_DIR}/upstream/zlib)
target_link_libraries(mobileudb_nodes PRIVATE zlibstatic)
if(MSVC)
  target_compile_definitions(mobileudb_nodes PRIVATE _CRT_SECURE_NO_WARNINGS stricmp=_stricmp strnicmp=_strnicmp)
else()
  target_compile_definitions(mobileudb_nodes PRIVATE stricmp=strcasecmp strnicmp=strncasecmp)
  target_compile_options(mobileudb_nodes PRIVATE -fvisibility=hidden)
endif()
if(ANDROID)
  target_link_options(mobileudb_nodes PRIVATE "-Wl,-z,max-page-size=16384")
endif()
